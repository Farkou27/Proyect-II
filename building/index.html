    🔨 Compilación y Ejecución¶

    Guía completa de opciones de build, configuraciones y troubleshooting.

    🖥️ Requisitos del Sistema¶

    Componente Mínimo Recomendado
    GCC 7.0+ 11.0+
    Clang 5.0+ 14.0+
    MSVC 19.1+ (VS 2017) 19.3+ (VS 2022)
    CMake 3.10+ 3.20+
    Estándar C++17 C++20

    ⚡ Compilación Rápida (GCC/Clang)¶

    Básico¶

    g++ -std=c++17 main.cpp Biblioteca.cpp -o biblioteca

    Con Optimizaciones (Release)¶

    g++ -std=c++17 -O3 -DNDEBUG main.cpp Biblioteca.cpp -o biblioteca

    Con Debug (Desarrollo)¶

    g++ -std=c++17 -g -O0 -Wall -Wextra -Wpedantic main.cpp Biblioteca.cpp -o biblioteca-debug

    Con Sanitizers (Detección de bugs)¶

    # AddressSanitizer - memory errors
g++ -std=c++17 -g -fsanitize=address -fno-omit-frame-pointer main.cpp Biblioteca.cpp -o biblioteca-asan

# UndefinedBehaviorSanitizer - UB detection
g++ -std=c++17 -g -fsanitize=undefined main.cpp Biblioteca.cpp -o biblioteca-ubsan

# ThreadSanitizer - data races (para futuro código concurrente)
g++ -std=c++17 -g -fsanitize=thread main.cpp Biblioteca.cpp -o biblioteca-tsan

    🏗️ Compilación con CMake (Recomendado)¶

    CMakeLists.txt¶

    cmake_minimum_required(VERSION 3.10)
project(BibliotecaCPP VERSION 2.0.0 LANGUAGES CXX)

# Estándar C++17
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Configuraciones por defecto
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

# Flags de compilación
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -Wextra -Wpedantic")
set(CMAKE_CXX_FLAGS_DEBUG "-g -O0")
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")

# Ejecutable principal
add_executable(biblioteca
    main.cpp
    Biblioteca.cpp
)

# Propiedades del target
target_compile_features(biblioteca PRIVATE cxx_std_17)

# Instalación (opcional)
install(TARGETS biblioteca DESTINATION bin)

# Testing (si añades tests)
enable_testing()
# add_subdirectory(tests)

    Build Out-of-Source¶

    # Configurar
mkdir build && cd build
cmake .. -DCMAKE_BUILD_TYPE=Release

# Compilar
make -j$(nproc)  # Paralelo usando todos los cores

# Ejecutar
./biblioteca

# Instalar (opcional)
sudo make install

    Build Types¶

    # Debug
cmake .. -DCMAKE_BUILD_TYPE=Debug

# Release
cmake .. -DCMAKE_BUILD_TYPE=Release

# RelWithDebInfo (optimizado con debug info)
cmake .. -DCMAKE_BUILD_TYPE=RelWithDebInfo

# MinSizeRelease (optimizado por tamaño)
cmake .. -DCMAKE_BUILD_TYPE=MinSizeRelease

    🪟 Windows (MSVC / Visual Studio)¶

    Developer Command Prompt¶

    cl /std:c++17 /EHsc /O2 main.cpp Biblioteca.cpp /Fe:biblioteca.exe

    Con CMake + Visual Studio¶

    mkdir build
cd build
cmake .. -G "Visual Studio 17 2022" -A x64
cmake --build . --config Release

    Windows + WSL (Recomendado)¶

    # En WSL (Ubuntu/Debian)
sudo apt update && sudo apt install build-essential cmake
# Luego seguir instrucciones Linux

    🐳 Docker (Entorno Reproducible)¶

    Dockerfile¶

    FROM gcc:13 AS builder
WORKDIR /app
COPY . .
RUN g++ -std=c++17 -O3 -DNDEBUG main.cpp Biblioteca.cpp -o biblioteca

FROM debian:bookworm-slim
WORKDIR /app
COPY --from=builder /app/biblioteca .
CMD ["./biblioteca"]

    Build y Run¶

    docker build -t biblioteca-cpp .
docker run --rm -it biblioteca-cpp

    📦 Empaquetado¶

    Debian/Ubuntu (.deb)¶

    # Estructura
mkdir -p pkg/DEBIAN pkg/usr/bin
cp biblioteca pkg/usr/bin/
cat > pkg/DEBIAN/control << EOF
Package: biblioteca-cpp
Version: 2.0.0
Section: utils
Priority: optional
Architecture: amd64
Maintainer: Tu Nombre <email@ejemplo.com>
Description: Sistema de gestión de biblioteca en C++17
EOF
dpkg-deb --build pkg biblioteca-cpp_2.0.0_amd64.deb

    AppImage (Linux portable)¶

    # Requiere appimagetool
mkdir -p AppDir/usr/bin
cp biblioteca AppDir/usr/bin/
cat > AppDir/AppRun << 'EOF'
#!/bin/sh
exec "$APPDIR/usr/bin/biblioteca" "$@"
EOF
chmod +x AppDir/AppRun
appimagetool AppDir biblioteca-cpp-x86_64.AppImage

    🧪 Testing Automatizado¶

    Con CTest (CMake)¶

    # En CMakeLists.txt
enable_testing()
add_test(NAME biblioteca_smoke_test COMMAND biblioteca <<< "0")

    cd build
ctest --output-on-failure

    Script de Test Simple¶

    #!/bin/bash
# test_smoke.sh
set -e

echo "=== Test 1: Ejecución básica ==="
echo "0" | ./biblioteca | grep -q "Hasta luego"
echo "✅ Test 1 passed"

echo "=== Test 2: Préstamo/Devolución ==="
echo -e "3\n1984\n4\n1984\n0" | ./biblioteca | grep -q "Prestados: 0"
echo "✅ Test 2 passed"

echo "=== Todos los tests pasaron ==="

    🔍 Análisis Estático¶

    Clang-Tidy¶

    # Instalar: sudo apt install clang-tidy
clang-tidy main.cpp Biblioteca.cpp -- -std=c++17

    Cppcheck¶

    # Instalar: sudo apt install cppcheck
cppcheck --enable=all --std=c++17 --suppress=missingIncludeSystem .

    Include What You Use (IWYU)¶

    # Instalar: sudo apt install iwyu
iwyu -Xiwyu --mapping_file=/dev/null main.cpp Biblioteca.cpp

    📊 Profiling y Performance¶

    Perf (Linux)¶

    # Compilar con debug info
g++ -std=c++17 -g -O3 main.cpp Biblioteca.cpp -o biblioteca

# Profilear
perf record -g ./biblioteca <<< "2"
perf report

    Valgrind (Memory)¶

    # Memcheck
valgrind --leak-check=full --show-leak-kinds=all ./biblioteca <<< "0"

# Callgrind (performance)
valgrind --tool=callgrind ./biblioteca <<< "2"
kcachegrind callgrind.out.*

    📋 Checklist de Release¶

    • Compila sin warnings (-Wall -Wextra -Wpedantic)
    • Pasa tests unitarios
    • Pasa AddressSanitizer sin errores
    • Pasa UndefinedBehaviorSanitizer sin errores
    • Valgrind sin memory leaks
    • Clang-tidy sin issues críticos
    • Documentación actualizada
    • Version bump (semver)
    • Changelog actualizado
    • Binario probado en distros objetivo

    ❌ Errores Comunes y Soluciones¶

    Error Causa Solución
    error: 'make_shared' is not a member of 'std' Estándar < C++11 Usar -std=c++17
    undefined reference to 'std::__cxx11::basic_string' ABI mismatch Recompilar todo con mismo GCC
    terminate called after throwing std::bad_alloc Sin memoria Optimizar o usar reserva
    cin >> n salta getline \n en buffer Usar cin.ignore() tras >>
    Caracteres raros en output Terminal sin UTF-8 export LANG=C.UTF-8
